\section*{רמזים}
\addcontentsline{toc}{section}{רמזים}
כאן מופיעים רק הרמזים, בלי הפתרונות. נסו קודם לפתור בעזרת הרמז, ורק אחר כך עברו לפתרון.


\subsection*{תורת הקבוצות ומאורעות}
\noindent \bknum{7}\ (א)\enskip לכל אזור צבוע, בדקו אם הוא בתוך כל אחת מהקבוצות $A,B,C$ או מחוץ לה.\par\smallskip
\noindent \bknum{8}\ (א)\enskip הכלה דו-כיוונית: הראו ש-$A\cap B\subseteq A$ וגם ש-$A\subseteq A\cap B$. בכיוון השני השתמשו בהנחה $A\subseteq B$.\par\smallskip
\noindent \bknum{9}\enskip כתבו את $A\cup B$ כאיחוד של שתי קבוצות זרות, $A\setminus B$ ו-$B$. לגודל של $A\setminus B$ השתמשו בעקרון החיסור.\par\smallskip
\noindent \bknum{11}\ (ב)\enskip תתי-הקבוצות של קבוצה עם איבר אחד הן הקבוצה הריקה והקבוצה עצמה.\par\smallskip

\subsection*{קומבינטוריקה}
\noindent \bknum{12}\ (ג)\enskip חשבו על כל נושא כעל "בלוק" אחד: קודם מסדרים את הבלוקים, ואחר כך את הספרים בתוך כל בלוק.\par\smallskip
\noindent \bknum{12}\ (ה)\enskip בכל שבוע בוחרים זוג ספרים, בלי חשיבות לסדר בתוך הזוג. הרשימה היא סדרה של 5 זוגות כאלה, אחד לכל שבוע, ולכן אפשר להשתמש בעקרון הכפל.\par\smallskip
\noindent \bknum{14}\ (ה)\enskip במספר כזה יש ספרה $X$ שמופיעה 3 פעמים, וספרה אחרת $Y$ שמופיעה פעם אחת. כמה אפשרויות יש לבחירת $X$, לבחירת $Y$, ולמקום של $Y$?\par\smallskip
\noindent \bknum{15}\ (ב)\enskip כמה סידורים של ספרים שונים נראים בדיוק אותו דבר, כשמסתכלים רק על הצבעים?\par\smallskip
\noindent \bknum{16}\ (ג)\enskip חשבו על שתי חנויות ההלבשה וחנות הנעליים שביניהן כעל "בלוק" של 3 מקומות רצופים.\par\smallskip
\noindent \bknum{22}\ (ד)\enskip קל יותר לספור את המשלים: בכמה דרכים קומה 4 \emph{לא} מיוצגת?\par\smallskip
\noindent \bknum{23}\ (א)\enskip כתבו את 6 כשורה של שש אחדות: $1\ \_\ 1\ \_\ 1\ \_\ 1\ \_\ 1\ \_\ 1$. בין האחדות יש 5 רווחים, ובכל רווח מחליטים אם לשים בו "$+$" או להשאיר אותו ריק.\par\smallskip
\noindent \bknum{25}\enskip גפן צריכה לארוז 13 זוגות מכנסיים לחופשה. בארון יש לה 30 זוגות, מתוכם 10 ארוכים ו-20 קצרים. בכמה דרכים היא יכולה לעשות זאת?\par\smallskip
\noindent \bknum{26}\enskip סמנו $|\Omega|=n$, ובחרו ערכים מתאימים ל-$a$ ול-$b$ בבינום של ניוטון.\par\smallskip
\noindent \bknum{27}\enskip כמה זוגות מכנסיים יש עכשיו לגפן בארון, וכמה היא צריכה לארוז?\par\smallskip
\noindent \bknum{28}\ (א)\enskip מתוך $n$ תלמידים, שאחד מהם הוא דני, בוחרים $k$. מפרידים לשני מקרים: דני נבחר, או דני לא נבחר.\par\smallskip
\noindent \bknum{29}\ (ב)\enskip אם ב-$n$ המקומות הראשונים יש $i$ אפסים, כמה אפסים יש בסדרה כולה?\par\smallskip
\noindent \bknum{30}\ (ד)\enskip קל יותר לספור את הקבוצות שבהן שניהם נמצאים.\par\smallskip
\noindent \bknum{31}\enskip כמה תתי-קבוצות \emph{לא} מקיימות את התנאי?\par\smallskip
\noindent \bknum{32}\ (ז)\enskip כל מחובר בפיתוח הוא מהצורה $\binom6k(4x)^k3^{6-k}$.\par\smallskip
\noindent \bknum{32}\ (ח)\enskip כל מחובר בפיתוח הוא מהצורה $\binom5k(a^2)^k(b^3)^{5-k}$. לאיזה $k$ מקבלים $a^6b^6$?\par\smallskip
\noindent \bknum{33}\ (א)\enskip מתוך $n$ סטודנטים בוחרים ועדה של $r$ חברים, ומתוך הוועדה בוחרים ועדת משנה של $k$ חברים. בכמה דרכים אפשר לעשות את זה, אם מתחילים מהוועדה? ואם מתחילים מוועדת המשנה?\par\smallskip
\noindent \bknum{34}\ (א)\enskip מתוך $n$ סטודנטים בוחרים ועדה (בכל גודל), ובוחרים לה יו"ר מבין חבריה.\par\smallskip
\noindent \bknum{34}\ (ב)\enskip בכיתה $n$ בנים ו-$n$ בנות, ובוחרים שני תלמידים.\par\smallskip

\subsection*{מבוא להסתברות}
\noindent \bknum{38}\enskip קודם בחרו את שני החודשים. בכמה דרכים אפשר לחלק את 6 האנשים בין שני החודשים, כך שאף חודש לא יישאר ריק?\par\smallskip
\noindent \bknum{39}\ (ב)\enskip חשבו על שני הספרים כעל „גוש” אחד.\par\smallskip
\noindent \bknum{40}\ (ב)\enskip האם המאורעות $\set{1,2},\set{2,3},\set{3,4},\dots$ זרים? חפשו מאורעות מהצורה $\set{k,k+1}$ שכן זרים בזוגות, ושהאיחוד שלהם הוא $\Omega$.\par\smallskip
\noindent \bknum{44}\ (ג)\enskip סמנו $A_i$ = "לסניף ה-$i$ הגיעה לפחות מדענית מחשב אחת", ועברו למשלים: $\PP(A_1\cap A_2\cap A_3)=1-\PP(A_1^C\cup A_2^C\cup A_3^C)$.\par\smallskip
\noindent \bknum{47}\ (א)\enskip סמנו $A$ = „עבר את הראשון”, $B$ = „עבר את השני”. אפשר גם לסדר את הנתונים בטבלה של $2\times2$: עבר/נכשל בראשון, ועבר/נכשל בשני.\par\smallskip
