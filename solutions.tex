\section*{רמזים}
\addcontentsline{toc}{section}{רמזים}
כאן מופיעים רק הרמזים, בלי הפתרונות. נסו קודם לפתור בעזרת הרמז, ורק אחר כך עברו לפתרון.


\subsection*{תורת הקבוצות ומאורעות}
\noindent \bknum{7}\ (א)\enskip לכל אזור צבוע, בדקו אם הוא בתוך כל אחת מהקבוצות $A,B,C$ או מחוץ לה.\par\smallskip
\noindent \bknum{8}\ (א)\enskip הכלה דו-כיוונית: הראו ש-$A\cap B\subseteq A$ וגם ש-$A\subseteq A\cap B$. בכיוון השני השתמשו בהנחה $A\subseteq B$.\par\smallskip
\noindent \bknum{9}\enskip כתבו את $A\cup B$ כאיחוד של שתי קבוצות זרות, $A\setminus B$ ו-$B$. לגודל של $A\setminus B$ השתמשו בעקרון החיסור.\par\smallskip
\noindent \bknum{11}\ (ב)\enskip תתי-הקבוצות של קבוצה עם איבר אחד הן הקבוצה הריקה והקבוצה עצמה.\par\smallskip

\subsection*{קומבינטוריקה}
\noindent \bknum{12}\ (ג)\enskip חשבו על כל נושא כעל "בלוק" אחד: קודם מסדרים את הבלוקים, ואחר כך את הספרים בתוך כל בלוק.\par\smallskip
\noindent \bknum{12}\ (ה)\enskip בכל שבוע בוחרים זוג ספרים, בלי חשיבות לסדר בתוך הזוג. הרשימה היא סדרה של 5 זוגות כאלה, אחד לכל שבוע, ולכן אפשר להשתמש בעקרון הכפל.\par\smallskip
\noindent \bknum{14}\ (ה)\enskip במספר כזה יש ספרה $X$ שמופיעה 3 פעמים, וספרה אחרת $Y$ שמופיעה פעם אחת. כמה אפשרויות יש לבחירת $X$, לבחירת $Y$, ולמקום של $Y$?\par\smallskip
\noindent \bknum{15}\ (ב)\enskip כמה סידורים של ספרים שונים נראים בדיוק אותו דבר, כשמסתכלים רק על הצבעים?\par\smallskip
\noindent \bknum{16}\ (ג)\enskip חשבו על שתי חנויות ההלבשה וחנות הנעליים שביניהן כעל "בלוק" של 3 מקומות רצופים.\par\smallskip
\noindent \bknum{21}\ (א)\enskip כתבו את 6 כשורה של שש אחדות: $1\ \_\ 1\ \_\ 1\ \_\ 1\ \_\ 1\ \_\ 1$. בין האחדות יש 5 רווחים, ובכל רווח מחליטים אם לשים בו "$+$" או להשאיר אותו ריק.\par\smallskip
