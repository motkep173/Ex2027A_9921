\section*{רמזים}
\addcontentsline{toc}{section}{רמזים}
כאן מופיעים רק הרמזים, בלי הפתרונות. נסו קודם לפתור בעזרת הרמז, ורק אחר כך עברו לפתרון.


\subsection*{תורת הקבוצות ומאורעות}
\noindent \bknum{5}\ (א)\enskip לכל אזור צבוע, בדקו אם הוא בתוך כל אחת מהקבוצות $A,B,C$ או מחוץ לה.\par\smallskip
\noindent \bknum{6}\ (א)\enskip הכלה דו-כיוונית: הראו ש-$A\cap B\subseteq A$ וגם ש-$A\subseteq A\cap B$. בכיוון השני השתמשו בהנחה $A\subseteq B$.\par\smallskip
\noindent \bknum{7}\enskip כתבו את $A\cup B$ כאיחוד של שתי קבוצות זרות, $A\setminus B$ ו-$B$. לגודל של $A\setminus B$ השתמשו בעקרון החיסור.\par\smallskip
\noindent \bknum{9}\ (ב)\enskip תתי-הקבוצות של קבוצה עם איבר אחד הן הקבוצה הריקה והקבוצה עצמה.\par\smallskip

\subsection*{קומבינטוריקה}
\noindent \bknum{10}\ (ג)\enskip חשבו על כל נושא כעל "בלוק" אחד: קודם מסדרים את הבלוקים, ואחר כך את הספרים בתוך כל בלוק.\par\smallskip
\noindent \bknum{10}\ (ה)\enskip בכל שבוע בוחרים זוג ספרים, בלי חשיבות לסדר בתוך הזוג. הרשימה היא סדרה של 5 זוגות כאלה, אחד לכל שבוע, ולכן אפשר להשתמש בעקרון הכפל.\par\smallskip
\noindent \bknum{15}\enskip סמנו $|\Omega|=n$, ובחרו ערכים מתאימים ל-$a$ ול-$b$ בבינום של ניוטון.\par\smallskip
\noindent \bknum{16}\enskip גפן צריכה לארוז 13 זוגות מכנסיים לחופשה. בארון יש לה 30 זוגות, מתוכם 10 ארוכים ו-20 קצרים. בכמה דרכים היא יכולה לעשות זאת?\par\smallskip
\noindent \bknum{17}\enskip כמה זוגות מכנסיים יש עכשיו לגפן בארון, וכמה היא צריכה לארוז?\par\smallskip
\noindent \bknum{18}\ (א)\enskip מתוך $n$ תלמידים, שאחד מהם הוא דני, בוחרים $k$. מפרידים לשני מקרים: דני נבחר, או דני לא נבחר.\par\smallskip
\noindent \bknum{19}\ (א)\enskip כתבו את 6 כשורה של שש אחדות: $1\ \_\ 1\ \_\ 1\ \_\ 1\ \_\ 1\ \_\ 1$. בין האחדות יש 5 רווחים, ובכל רווח מחליטים אם לשים בו "$+$" או להשאיר אותו ריק.\par\smallskip

\subsection*{מבוא להסתברות}
\noindent \bknum{21}\ (ג)\enskip סמנו $A_i$ = "לסניף ה-$i$ הגיעה לפחות מדענית מחשב אחת", ועברו למשלים: $\PP(A_1\cap A_2\cap A_3)=1-\PP(A_1^C\cup A_2^C\cup A_3^C)$.\par\smallskip
\noindent \bknum{22}\ (ב)\enskip האם המאורעות $\set{1,2},\set{2,3},\set{3,4},\dots$ זרים? חפשו מאורעות מהצורה $\set{k,k+1}$ שכן זרים בזוגות, ושהאיחוד שלהם הוא $\Omega$.\par\smallskip

\subsection*{הסתברות מותנית ואי-תלות}
\noindent \bknum{24}\ (א)\enskip השתמשו בנוסחת ההסתברות השלמה, עם חלוקה לפי הקופסה שהפרופסור בחר.\par\smallskip
\noindent \bknum{24}\ (ב)\enskip השתמשו בנוסחת בייס ובתשובה לסעיף הקודם.\par\smallskip

\subsection*{משתנים מקריים}
\noindent \bknum{25}\enskip סמנו $p_k=\PP(X=k)$ וכתבו כל אחד מהנתונים כמשוואה, למשל $F_X(2)=p_0+p_1+p_2$. אל תשכחו ש-$p_0+p_1+p_2+p_3+p_4=1$.\par\smallskip
\noindent \bknum{26}\enskip $X=k$ כשהסופגניות שנדב בוחר בפעמים $1,\dots,k-1$ אינן שוקולד, והסופגנייה שהוא בוחר בפעם ה-$k$ היא שוקולד. כפלו את ההסתברויות המותנות של כל שלב.\par\smallskip
\noindent \bknum{27}\ (א)\enskip מה צריך לקרות בשלוש השנים הראשונות, ומה בשנה הרביעית?\par\smallskip
\noindent \bknum{27}\ (ד)\enskip בכל שנה יש ארבע אפשרויות לשני הקורסים. בסעיף השני חלקו לפי השנה $k$ שבה שניהם יוצאים לשבתון, וסכמו על $k$.\par\smallskip
\noindent \bknum{28}\ (א)\enskip מתי $X+Y=1$? שני המשתנים מקבלים רק ערכים שלמים ואי-שליליים.\par\smallskip
\noindent \bknum{28}\ (ג)\enskip חלקו לפי הערך של $X$, שיכול להיות רק $0,1,2,3$ או $4$.\par\smallskip

\subsection*{תוחלת ושונות}
\noindent \bknum{30}\ (ב)\enskip רשמו את כל הערכים האפשריים של $Y$ ואת ההסתברות של כל אחד מהם. לשונות היעזרו ב-$\Var[Y]=\E[Y^2]-(\E[Y])^2$.\par\smallskip
\noindent \bknum{31}\ (ב)\enskip סמנו $p_k=\PP(X=k)$ וכתבו שלוש משוואות: סכום ההסתברויות, התוחלת, ו-$\E[X^2]=\Var[X]+(\E[X])^2$.\par\smallskip
\noindent \bknum{32}\ (א)\enskip לכל קורס $i$ הגדירו משתנה $I_i$, ששווה ל-1 אם אף סטודנט מהמחלקה לא בחר בקורס, ול-0 אחרת. מספר הקורסים הריקים הוא $I_1+\dots+I_{100}$.\par\smallskip
