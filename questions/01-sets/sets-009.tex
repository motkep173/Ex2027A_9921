\begin{enumerate}
\item תהי $A=\set{1,2,3}$. כתבו את $\Pow(A)$, ובדקו ש-$|\Pow(A)|=2^{|A|}$.
\begin{solbox}
$\Pow(A)=\ans{\set{\emptyset,\ \set1,\ \set2,\ \set3,\ \set{1,2},\ \set{1,3},\ \set{2,3},\ \set{1,2,3}}}$. יש בה 8 איברים, ואכן $2^{|A|}=2^3=8$.
\end{solbox}
\item מצאו את $\Pow(\emptyset)$ ואת $\Pow(\Pow(\emptyset))$, ואת הגודל של כל אחת מהן.
\hint{תתי-הקבוצות של קבוצה עם איבר אחד הן הקבוצה הריקה והקבוצה עצמה.}
\begin{solbox}
לקבוצה הריקה יש תת-קבוצה אחת בלבד, היא עצמה, ולכן $\Pow(\emptyset)=\ans{\set{\emptyset}}$, קבוצה עם איבר אחד. ל-$\set\emptyset$ יש שתי תתי-קבוצות, $\emptyset$ ו-$\set\emptyset$, ולכן $\Pow(\Pow(\emptyset))=\ans{\set{\emptyset,\set\emptyset}}$, עם 2 איברים. זה מתאים ל-$2^0=1$ ול-$2^1=2$.
\end{solbox}
\item תהי $A$ קבוצה סופית. מהי $A\times\emptyset$? האם זה מתאים לעקרון הכפל?
\begin{solbox}
אין אף זוג $(a,b)$ עם $b\in\emptyset$, ולכן $A\times\emptyset=\ans{\emptyset}$. זה מתאים לעקרון הכפל: $|A\times\emptyset|=|A|\cdot0=0$.
\end{solbox}
\end{enumerate}
