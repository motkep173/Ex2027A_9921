בסעיפים א--ג כתבו את הקבוצה ברשימת איברים.
\begin{enumerate}
\item $A=\setof{x\in\N}{x\le5}$.
\begin{solbox}
$A=\ans{\set{1,2,3,4,5}}$.
\end{solbox}
\item $B=\setof{2k+1}{k\in\set{0,1,2,3}}$.
\begin{solbox}
מציבים $k=0,1,2,3$ ומקבלים $1,3,5,7$, ולכן $B=\ans{\set{1,3,5,7}}$.
\end{solbox}
\item $C=\setof{x\in\mathbb{Z}}{x^2\le4}$.
\begin{solbox}
$x^2\le4$ בדיוק כאשר $-2\le x\le2$. המספרים השלמים בקטע הזה הם $C=\ans{\set{-2,-1,0,1,2}}$.
\end{solbox}
\item ולהפך: כתבו את הקבוצה $D=\set{3,6,9,12,\dots}$ בעזרת תנאי.
\begin{solbox}
אלה הכפולות של 3: $D=\ans{\setof{3n}{n\in\N}}$.
\end{solbox}
\item ולהפך: כתבו את הקבוצה $E=\set{1,4,9,16,25}$ בעזרת תנאי.
\begin{solbox}
אלה ריבועי המספרים $1,\dots,5$: $E=\ans{\setof{n^2}{n\in\N,\ n\le5}}$.
\end{solbox}
\end{enumerate}
