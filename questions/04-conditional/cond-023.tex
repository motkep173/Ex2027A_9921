יהיו $A$ ו־$B$ מאורעות במרחב הסתברות, כך ש־$\PP(A\cup B)=0.8$ ו־$\PP(A^C)=0.7$. מצאו את $\PP(B)$ בכל אחד מהמקרים הבאים.
\begin{enumerate}
\item $A$ ו־$B$ זרים.
\begin{solbox}
$\PP(A)=0.3$. כשהמאורעות זרים, $\PP(A\cup B)=\PP(A)+\PP(B)$, ולכן $\PP(B)=0.8-0.3=\ans{0.5}$.
\end{solbox}
\item $A$ ו־$B$ בלתי תלויים.
\begin{solbox}
נסמן $p=\PP(B)$. אז $\PP(A\cap B)=0.3p$, ולפי עקרון ההכלה וההפרדה $0.8=0.3+p-0.3p$. מכאן $0.7p=0.5$, כלומר $p=\ans{\frac57}$.
\end{solbox}
\item $\PP(A\mid B)=0.2$.
\begin{solbox}
נסמן $p=\PP(B)$. לפי כלל הכפל $\PP(A\cap B)=\PP(A\mid B)\PP(B)=0.2p$, ולכן $0.8=0.3+p-0.2p$. מכאן $0.8p=0.5$, כלומר $p=\ans{\frac58}$.
\end{solbox}
\end{enumerate}
