יהיו $A$ ו־$B$ מאורעות במרחב הסתברות, כך ש־$\PP(A)=0.4$, $\PP(B)=0.25$ ו־$\PP(A\cap B)=0.15$. חשבו:
\begin{enumerate}
\item $\PP(A^C\cap B^C)$
\begin{solbox}
לפי כללי דה-מורגן $A^C\cap B^C=(A\cup B)^C$. לפי הכלה והפרדה $\PP(A\cup B)=0.4+0.25-0.15=0.5$, ולכן
$\PP(A^C\cap B^C)=1-0.5=\ans{0.5}$.
\end{solbox}
\item $\PP(B\mid A)$
\begin{solbox}
$\PP(B\mid A)=\frac{\PP(A\cap B)}{\PP(A)}=\frac{0.15}{0.4}=\ans{\frac38}$.
\end{solbox}
\item $\PP(A^C\mid B)$
\begin{solbox}
$B$ הוא האיחוד הזר של $A\cap B$ ו-$A^C\cap B$, ולכן $\PP(A^C\cap B)=0.25-0.15=0.1$, ו-
\[
\PP(A^C\mid B)=\frac{0.1}{0.25}=\ans{0.4}.
\]
דרך אחרת: $\PP(A^C\mid B)=1-\PP(A\mid B)=1-\frac{0.15}{0.25}=0.4$.
\end{solbox}
\end{enumerate}
