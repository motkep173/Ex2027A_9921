נסמן $\Omega=\set{A,B,C,D,E,F,G,H}$.
\begin{enumerate}
\item כמה תתי-קבוצות שונות בגודל 2 יש ל-$\Omega$?
\begin{solbox}
\unverified
בוחרים שני איברים שונים מתוך 8, בלי חשיבות לסדר ובלי חזרות: $\binom82=\frac{7\cdot8}2=\ans{28}$.
\end{solbox}
\item כמה תתי-קבוצות שונות יש ל-$\Omega$?
\begin{solbox}
\unverified
$2^{|\Omega|}=2^8=\ans{256}$.
\end{solbox}
\item כמה מהן בגודל זוגי?
\begin{solbox}
\unverified
תת-קבוצה בגודל זוגי היא מגודל $0,2,4,6$ או $8$, ולכן מספרן
\[
\binom80+\binom82+\binom84+\binom86+\binom88=1+28+70+28+1=\ans{128}.
\]
\end{solbox}
\item כמה מהן בגודל אי-זוגי?
\begin{solbox}
\unverified
באותו אופן, $\binom81+\binom83+\binom85+\binom87=8+56+56+8=\ans{128}$.
\end{solbox}
\end{enumerate}
