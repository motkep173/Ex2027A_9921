היעזרו בבינום של ניוטון.
\begin{enumerate}
\item כתבו במפורש את $(1+x)^8$.
\begin{solbox}
לפי הבינום של ניוטון, $(1+x)^8=\sum_{k=0}^8\binom8kx^k$:
\[
(1+x)^8=1+8x+28x^2+56x^3+70x^4+56x^5+28x^6+8x^7+x^8.
\]
\end{solbox}
\item כתבו במפורש את $(1-x)^8$.
\begin{solbox}
מציבים $-x$ במקום $x$ בסעיף הקודם. המקדמים של החזקות האי-זוגיות מחליפים סימן:
\[
(1-x)^8=1-8x+28x^2-56x^3+70x^4-56x^5+28x^6-8x^7+x^8.
\]
\end{solbox}
\item מצאו את המקדם של $x^5$ ב-$(1+x)^{11}$.
\begin{solbox}
$\binom{11}5=\ans{462}$.
\end{solbox}
\item מצאו את המקדם של $a^2b^8$ ב-$(a+b)^{10}$.
\begin{solbox}
$\binom{10}2=\ans{45}$.
\end{solbox}
\item מצאו את המקדם של $x^6$ ב-$(1+x)^{12}$.
\begin{solbox}
$\binom{12}6=\ans{924}$.
\end{solbox}
\item מצאו את המקדם של $a^3b^6$ ב-$(a+b)^9$.
\begin{solbox}
$\binom93=\ans{84}$.
\end{solbox}
\item מצאו את המקדם של $x^3$ ב-$(3+4x)^6$.
\hint{כל מחובר בפיתוח הוא מהצורה $\binom6k(4x)^k3^{6-k}$.}
\begin{solbox}
המחובר הכללי הוא $\binom6k(4x)^k3^{6-k}$. עבור $k=3$: $\binom63\cdot4^3\cdot3^3=20\cdot64\cdot27=\ans{34{,}560}$.
\end{solbox}
\item מצאו את המקדם של $a^6b^6$ ב-$(a^2+b^3)^5$.
\hint{כל מחובר בפיתוח הוא מהצורה $\binom5k(a^2)^k(b^3)^{5-k}$. לאיזה $k$ מקבלים $a^6b^6$?}
\begin{solbox}
המחובר הכללי הוא $\binom5k(a^2)^k(b^3)^{5-k}=\binom5ka^{2k}b^{15-3k}$. כדי לקבל $a^6b^6$ צריך $2k=6$ ו-$15-3k=6$, כלומר $k=3$. המקדם הוא $\binom53=\ans{10}$.
\end{solbox}
\end{enumerate}
