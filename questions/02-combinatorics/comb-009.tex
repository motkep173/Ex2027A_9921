נסמן $\Omega=\set{A,B,C}$.
\begin{enumerate}
\item כמה תתי-קבוצות שונות יש ל-$\Omega$? כתבו את כולן.
\begin{solbox}
\unverified
יש $2^3=\ans8$ תתי-קבוצות: $\emptyset,\ \set A,\ \set B,\ \set C,\ \set{A,B},\ \set{A,C},\ \set{B,C},\ \set{A,B,C}$.
\end{solbox}
\item כמה מהן בגודל זוגי? ($0$ נחשב זוגי.)
\begin{solbox}
\unverified
יש בדיוק $\ans4$ בגודל זוגי: $\emptyset,\ \set{A,B},\ \set{A,C},\ \set{B,C}$.
\end{solbox}
\item כמה מהן בגודל אי-זוגי?
\begin{solbox}
\unverified
יש בדיוק $\ans4$ בגודל אי-זוגי: $\set A,\ \set B,\ \set C,\ \set{A,B,C}$.
\end{solbox}
\end{enumerate}
