\textbf{חשבו ללא מחשבון:}
\begin{enumerate}
\item $\binom42$
\begin{solbox}
\unverified
$\binom42=\frac{4!}{2!\,2!}=\frac{3\cdot4}{2}=\ans6$.
\end{solbox}
\item $\binom{100}1$
\begin{solbox}
\unverified
יש בדיוק 100 דרכים לבחור עצם אחד מתוך 100, ולכן $\binom{100}1=\ans{100}$. גם מהנוסחה: $\binom{100}1=\frac{100!}{1!\,99!}=100$.
\end{solbox}
\item $\binom{20}2$
\begin{solbox}
\unverified
$\binom{20}2=\frac{20!}{2!\,18!}=\frac{19\cdot20}2=\ans{190}$.
\end{solbox}
\end{enumerate}
