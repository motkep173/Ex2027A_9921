סתיו מסדרת באקראי 10 ספרים שונים על מדף.
\begin{enumerate}
\item מה ההסתברות שהספרים יהיו מסודרים לפי סדר האלף-בית של שמותיהם?
\begin{solbox}
מרחב המדגם הוא כל הסידורים של 10 הספרים, $|\Omega|=10!$, כולם סבירים באותה מידה. רק סידור אחד הוא לפי סדר האלף-בית, ולכן ההסתברות היא $\ans{\frac1{10!}}$.
\end{solbox}
\item שניים מהספרים הם „הנסיך הקטן” ו„פו הדוב”. מה ההסתברות שהם יעמדו זה ליד זה?
\hint{חשבו על שני הספרים כעל „גוש” אחד.}
\begin{solbox}
נדביק את שני הספרים ל„גוש” אחד. יש $9!$ דרכים לסדר את הגוש ואת 8 הספרים האחרים, ו-2 דרכים לסדר את שני הספרים בתוך הגוש. לכן ההסתברות היא $\frac{2\cdot9!}{10!}=\frac2{10}=\ans{\frac15}$.
\end{solbox}
\item מה ההסתברות שבין שני הספרים האלה יהיו בדיוק 3 ספרים?
\begin{solbox}
אם הספר השמאלי מבין השניים במקום $i$, השני במקום $i+4$, ולכן $1\le i\le6$: 6 אפשרויות למקומות. יש 2 דרכים להחליט מי משני הספרים בצד שמאל, ו-$8!$ דרכים לסדר את שאר הספרים. לכן ההסתברות היא
\[
\frac{6\cdot2\cdot8!}{10!}=\frac{12}{90}=\ans{\frac2{15}}.
\]
\end{solbox}
\item מבין 10 הספרים, ל-4 יש כריכה אדומה, ל-4 כריכה כחולה ול-2 כריכה לבנה. מה ההסתברות שהספרים מכל צבע יעמדו יחד?
\begin{solbox}
את שלושת „גושי הצבע” מסדרים ב-$3!$ דרכים, ובתוך כל גוש מסדרים את הספרים: $4!\cdot4!\cdot2!$. לכן ההסתברות היא
\[
\frac{3!\cdot4!\cdot4!\cdot2!}{10!}=\frac{6912}{3{,}628{,}800}=\ans{\frac1{525}}.
\]
\end{solbox}
\end{enumerate}
