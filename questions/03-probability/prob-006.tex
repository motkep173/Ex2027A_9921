מטילים קובייה הוגנת.
\begin{enumerate}
\item כתבו את מרחב המדגם $\Omega$.
\begin{solbox}
$\Omega=\set{1,2,3,4,5,6}$. הקובייה הוגנת, ולכן כל התוצאות סבירות באותה מידה: זה מרחב הסתברות אחיד, ו-$\PP(A)=\frac{|A|}{|\Omega|}=\frac{|A|}6$ לכל מאורע $A$.
\end{solbox}
\item מה ההסתברות לקבל 6?
\begin{solbox}
$\PP(\set6)=\ans{\frac16}$.
\end{solbox}
\item כתבו את המאורע „התקבלה תוצאה זוגית” כתת-קבוצה של $\Omega$, ומצאו את ההסתברות שלו.
\begin{solbox}
$A=\set{2,4,6}$, ולכן $\PP(A)=\frac36=\ans{\frac12}$.
\end{solbox}
\end{enumerate}
