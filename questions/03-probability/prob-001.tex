בסיום כנס בנושא בינה מלאכותית (\textenglish{AI}), 10 מדעניות מחשב קבעו להיפגש בסניף ארומה שבאוניברסיטת בן-גוריון. להפתעתן, התברר שיש באוניברסיטה שלושה סניפים כאלה, ולכן כל אחת מהן בחרה סניף באקראי והגיעה אליו.
\begin{solbox}
מרחב המדגם הוא כל הדרכים שבהן 10 מדעניות המחשב בוחרות סניפים, $|\Omega|=3^{10}$, ומרחב ההסתברות אחיד.
\end{solbox}
\begin{enumerate}
\item מה ההסתברות שכל מדעניות המחשב הצליחו להיפגש כמתוכנן, כלומר הגיעו כולן לאותו סניף?
\begin{solbox}
נסמן $A$ = "כל מדעניות המחשב הגיעו לאותו סניף". $|A|=3$, כי כולן יכולות להגיע לסניף הראשון, לשני או לשלישי. לכן
\[
\PP(A)=\frac{|A|}{|\Omega|}=\frac3{3^{10}}=\ans{\frac1{3^9}}.
\]
\end{solbox}
\item מה ההסתברות שלסניף בבניין 56 לא הגיעה אף אחת?
\begin{solbox}
נסמן $B$ = "לסניף בבניין 56 לא הגיעה אף אחת". אז $|B|=2^{10}$, כי לכל מדענית מחשב יש רק 2 אפשרויות ולא 3. לכן
\[
\PP(B)=\frac{2^{10}}{3^{10}}=\ans{\Big(\frac23\Big)^{10}}.
\]
\end{solbox}
\item השתמשו בעקרון ההכלה וההפרדה, וחשבו את ההסתברות שלכל סניף תגיע לפחות מדענית מחשב אחת.
\hint{סמנו $A_i$ = "לסניף ה-$i$ הגיעה לפחות מדענית מחשב אחת", ועברו למשלים: $\PP(A_1\cap A_2\cap A_3)=1-\PP(A_1^C\cup A_2^C\cup A_3^C)$.}
\begin{solbox}
נסמן $A_i$ = "לסניף ה-$i$ הגיעה לפחות מדענית מחשב אחת", $i=1,2,3$. אנחנו מחפשים את $\PP(A_1\cap A_2\cap A_3)$. לפי המשלים ודה־מורגן:
\[
\PP(A_1\cap A_2\cap A_3)=1-\PP(A_1^C\cup A_2^C\cup A_3^C),
\]
ולפי עקרון ההכלה וההפרדה:
\[
\begin{split}
\PP(A_1^C\cup A_2^C\cup A_3^C)={}&\PP(A_1^C)+\PP(A_2^C)+\PP(A_3^C)\\
&-\PP(A_1^C\cap A_2^C)-\PP(A_1^C\cap A_3^C)-\PP(A_2^C\cap A_3^C)\\
&+\PP(A_1^C\cap A_2^C\cap A_3^C).
\end{split}
\]

\begin{itemize}
\item כל $A_i^C$ פירושו שלסניף ה-$i$ לא הגיעה אף אחת. כמו בסעיף (ב), $\PP(A_i^C)=(\frac23)^{10}$.
\item כל $A_i^C\cap A_j^C$ פירושו שכולן הגיעו לסניף הנותר, ולכן $\PP(A_i^C\cap A_j^C)=\frac1{3^{10}}$.
\item $A_1^C\cap A_2^C\cap A_3^C=\emptyset$, ולכן הסתברותו $0$.
\end{itemize}
לכן
\[
\PP(A_1\cap A_2\cap A_3)=1-\Big(3\cdot\Big(\frac23\Big)^{10}-3\cdot\Big(\frac13\Big)^{10}\Big)=\ans{\frac{3^9-2^{10}+1}{3^9}}\approx0.948.
\]
\end{solbox}
\end{enumerate}
